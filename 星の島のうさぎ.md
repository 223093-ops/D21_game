        function spawnCatOrder() {
            if (gameState.catOrders.length >= 3) return;

            const cat = CAT_CUSTOMERS[Math.floor(Math.random() * CAT_CUSTOMERS.length)];
            let possibleKeys = ['carrot', 'wheat', 'bread', 'carrot_juice', 'egg', 'ham'];
            if (gameState.level >= 2) possibleKeys.push('strawberry', 'milk', 'butter', 'sandwich', 'cake');

            const reqKey = possibleKeys[Math.floor(Math.random() * possibleKeys.length)];
            const reqAmount = Math.floor(Math.random() * 2) + 1;
            const itemDef = ITEMS[reqKey];

            const rewardCoins = (itemDef.sellPrice || 12) * reqAmount + 8;
            const rewardExp = (itemDef.exp || 4) * reqAmount + 4;

                id: Date.now() + Math.random(),
                cat: cat,
                itemKey: reqKey,
                amount: reqAmount,
                coins: rewardCoins,
                exp: rewardExp
            });

            renderCatOrders();
        }

        function fulfillOrder(orderId) {
            const idx = gameState.catOrders.findIndex(o => o.id == orderId);
            if (idx === -1) return;

            const order = gameState.catOrders[idx];
            if ((gameState.inventory[order.itemKey] || 0) >= order.amount) {
                gameState.inventory[order.itemKey] -= order.amount;
                addCoins(order.coins);
                addExp(order.exp);

                // Move rabbit towards dock to deliver
                playerRabbit.targetR = 1;
                playerRabbit.targetC = 1;
                playerRabbit.actionAnimTimer = 20;

                gameState.catOrders.splice(idx, 1);
                renderCatOrders();
                playSFX('coin');
                showToast(`+${order.coins}🪙 +${order.exp}EXP お届け完了！`, window.innerWidth/2, window.innerHeight/3);

                setTimeout(spawnCatOrder, 4000);
            }
        }

        function isoToScreen(r, c) {
            const x = (c - r) * (TILE_WIDTH / 2) * camera.zoom + camera.x;
            const y = (c + r) * (TILE_HEIGHT / 2) * camera.zoom + camera.y;
            return { x, y };
        }

        function screenToIso(sx, sy) {
            const x = (sx - camera.x) / camera.zoom;
            const y = (sy - camera.y) / camera.zoom;
            const c = (x / (TILE_WIDTH / 2) + y / (TILE_HEIGHT / 2)) / 2;
            const r = (y / (TILE_HEIGHT / 2) - x / (TILE_WIDTH / 2)) / 2;
            return { r: Math.round(r), c: Math.round(c) };
        }

        function drawTilePolygon(x, y, topColor, strokeColor) {
            const w = TILE_WIDTH * camera.zoom;
            const h = TILE_HEIGHT * camera.zoom;

            ctx.beginPath();
            ctx.moveTo(x, y);
            ctx.lineTo(x + w / 2, y + h / 2);
            ctx.lineTo(x, y + h);
            ctx.lineTo(x - w / 2, y + h / 2);
            ctx.closePath();

            ctx.fillStyle = topColor;
            ctx.fill();

            if (strokeColor) {
                ctx.strokeStyle = strokeColor;
                ctx.lineWidth = 1;
                ctx.stroke();
            }
        }

        let gameFrame = 0;

        function renderGame() {
            gameFrame++;
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Water ocean background
            ctx.fillStyle = '#38bdf8';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Update Player Rabbit Movement
            updatePlayerMovement();

            // Update Island Rabbit Friends AI
            updateRabbitFriendsAI();

            // Render Map Tiles
            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const tile = gameState.grid[r][c];
                    const { x, y } = isoToScreen(r, c);

                    if (tile.type === 'water') {
                        // Water tile
                        const wave = Math.sin(gameFrame * 0.05 + r + c) * 2;
                        drawTilePolygon(x, y + wave, '#0284c7', '#0369a1');

                        if (isWaterAdjacentToGrass(r, c)) {
                            ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
                            ctx.font = `bold ${10 * camera.zoom}px sans-serif`;
                            ctx.textAlign = 'center';
                            ctx.fillText('🌊 開拓', x, y + 16 * camera.zoom);
                        }
                        continue;
                    }

                    // Grass tile
                    const isEven = (r + c) % 2 === 0;
                    drawTilePolygon(x, y, isEven ? '#86efac' : '#4ade80', '#22c55e');

                    // Highlight selected tile
                    if (selectedTile && selectedTile.r === r && selectedTile.c === c) {
                        drawTilePolygon(x, y, 'rgba(254, 240, 138, 0.6)', '#f59e0b');
                    }

                    // Render Building / Soil Plot
                    if (tile.building === 'plot') {
                        const w = TILE_WIDTH * camera.zoom * 0.72;
                        const h = TILE_HEIGHT * camera.zoom * 0.72;
                        ctx.fillStyle = '#854d0e';
                        ctx.beginPath();
                        ctx.ellipse(x, y + h/2, w/2, h/2, 0, 0, Math.PI * 2);
                        ctx.fill();

                        if (tile.crop) {
                            const elapsed = (Date.now() - tile.crop.plantedAt) / 1000;

                            ctx.textAlign = 'center';

                                ctx.fillStyle = '#fbbf24';
                                ctx.beginPath();
                                ctx.arc(x + 12 * camera.zoom, y - 6 * camera.zoom, 5 * camera.zoom, 0, Math.PI * 2);
                                ctx.fill();
                            } else {
                                ctx.fillText('🌱', x, y + 10 * camera.zoom);
                            }
                        }
                    } else if (tile.building === 'cow_barn') {
                        // Render Cow Barn with Multiple Full-Body Cows
                        drawBarnStructure(x, y, '#fef08a', '#eab308');
                        drawFullBodyCow(x - 12 * camera.zoom, y + 2 * camera.zoom, 0.8 * camera.zoom, gameFrame, 1);

                    } else if (tile.building === 'pig_pen') {
                        // Render Pig Pen with Multiple Full-Body Pigs
                        drawBarnStructure(x, y, '#fbcfe8', '#ec4899');
                        drawFullBodyPig(x - 10 * camera.zoom, y + 5 * camera.zoom, 0.8 * camera.zoom, gameFrame, 1);

                        ctx.font = `${32 * camera.zoom}px sans-serif`;

                        ctx.textAlign = 'center';
                        ctx.font = `${32 * camera.zoom}px sans-serif`;
                        ctx.fillText('風', x, y + 12 * camera.zoom);

        function updatePlayerMovement() {
            if (playerRabbit.r !== playerRabbit.targetR || playerRabbit.c !== playerRabbit.targetC) {
                playerRabbit.isWalking = true;
                playerRabbit.animCycle += 0.25;

                const speed = 0.08;
                const dr = playerRabbit.targetR - playerRabbit.r;
                const dc = playerRabbit.targetC - playerRabbit.c;
                const dist = Math.hypot(dr, dc);

                if (dist < 0.1) {
                    playerRabbit.r = playerRabbit.targetR;
                    playerRabbit.c = playerRabbit.targetC;
                    playerRabbit.isWalking = false;
                } else {
                    playerRabbit.r += (dr / dist) * speed;
                    playerRabbit.c += (dc / dist) * speed;
                    playerRabbit.facing = (dc - dr) >= 0 ? 1 : -1;
                }
            } else {
                playerRabbit.isWalking = false;
                playerRabbit.animCycle += 0.04;
            }

            if (playerRabbit.actionAnimTimer > 0) playerRabbit.actionAnimTimer--;
        }

        function updateRabbitFriendsAI() {
            rabbitFriends.forEach(friend => {
                if (friend.r !== friend.targetR || friend.c !== friend.targetC) {
                    friend.isWalking = true;
                    friend.animCycle += 0.22;

                    const speed = 0.05;
                    const dr = friend.targetR - friend.r;
                    const dc = friend.targetC - friend.c;
                    const dist = Math.hypot(dr, dc);

                    if (dist < 0.1) {
                        friend.r = friend.targetR;
                        friend.c = friend.targetC;
                        friend.isWalking = false;
                        friend.idleTimer = 100 + Math.random() * 200;
                    } else {
                        friend.r += (dr / dist) * speed;
                        friend.c += (dc / dist) * speed;
                        friend.facing = (dc - dr) >= 0 ? 1 : -1;
                    }
                } else {
                    friend.isWalking = false;
                    friend.animCycle += 0.03;

                    if (friend.idleTimer > 0) {
                        friend.idleTimer--;
                    } else {
                        const nr = Math.max(1, Math.min(GRID_SIZE - 2, Math.floor(friend.r + (Math.random() * 5 - 2))));
                        const nc = Math.max(1, Math.min(GRID_SIZE - 2, Math.floor(friend.c + (Math.random() * 5 - 2))));
                        if (gameState.grid[nr] && gameState.grid[nr][nc] && gameState.grid[nr][nc].type === 'grass') {
                            friend.targetR = nr;
                            friend.targetC = nc;
                        }
                    }
                }
            });
        }

            const legAngle = isWalking ? Math.sin(animCycle * 2.5) * 20 : 0;
            const armAngle = isCelebrating ? -120 : (isWalking ? Math.cos(animCycle * 2.5) * 15 : Math.sin(animCycle) * 4);

            ctx.fill();
            ctx.strokeStyle = palette.stroke;
            ctx.lineWidth = 1;
            ctx.stroke();
            ctx.restore();

            ctx.fillStyle = palette.body;
            ctx.beginPath();
            ctx.ellipse(0, -17 - bounce, 12, 14, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.strokeStyle = palette.stroke;
            ctx.lineWidth = 1.2;
            ctx.stroke();

            // Belly Patch
            ctx.fillStyle = palette.innerEar;
            ctx.beginPath();
            ctx.ellipse(1, -16 - bounce, 6.5, 8.5, 0, 0, Math.PI * 2);
            ctx.fill();

        function isWaterAdjacentToGrass(r, c) {
            const dirs = [[-1,0],[1,0],[0,-1],[0,1]];
            for (let d of dirs) {
                const nr = r + d[0], nc = c + d[1];
                if (nr >= 0 && nr < GRID_SIZE && nc >= 0 && nc < GRID_SIZE) {
                    if (gameState.grid[nr][nc].type === 'grass') return true;
                }
            }
            return false;
        }

        canvas.addEventListener('click', (e) => {
            const rect = canvas.getBoundingClientRect();
            const clickX = e.clientX - rect.left;
            const clickY = e.clientY - rect.top;

            const iso = screenToIso(clickX, clickY);
            const r = iso.r, c = iso.c;

            if (r < 0 || r >= GRID_SIZE || c < 0 || c >= GRID_SIZE) {
                hideTilePopover();
                return;
            }

            const tile = gameState.grid[r][c];
            selectedTile = { r, c };

            playerRabbit.targetR = r;
            playerRabbit.targetC = c;
            playSFX('pop');

            // Building / Facility Placement Mode
            if (gameState.buildingToPlace) {
                if (tile.type === 'grass' && !tile.building) {
                    tile.building = gameState.buildingToPlace;
                    showToast(`✨ ${BUILDINGS[gameState.buildingToPlace].name}を設置しました！`, clickX, clickY);
                    playerRabbit.actionAnimTimer = 25;
                    gameState.buildingToPlace = null;
                    hideTilePopover();
                    return;
                } else {
                    showToast(`⚠️ ここには建てられません`, clickX, clickY);
                    gameState.buildingToPlace = null;
                    return;
                }
            }

            // Water Tile Expansion
            if (tile.type === 'water') {
                if (isWaterAdjacentToGrass(r, c)) {
                    showTilePopover(e.clientX, e.clientY, `
                        <div class="text-xs font-black text-amber-950 mb-1">🌊 島を開拓しますか？</div>
                        <button onclick="expandWaterTile(${r}, ${c})" class="bg-gradient-to-r from-amber-400 to-yellow-400 hover:from-amber-500 hover:to-yellow-500 text-amber-950 font-black px-3 py-1.5 rounded-xl text-xs shadow border border-yellow-300 transition active:scale-95">
                            開拓する (80🪙)
                        </button>
                    `);
                } else {
                    hideTilePopover();
                }
                return;
            }

            // Direct Soil & Crop Actions (Empty Grass OR Soil Plot)
            if (tile.type === 'grass' && (!tile.building || tile.building === 'plot')) {
                if (tile.crop) {
                    const elapsed = (Date.now() - tile.crop.plantedAt) / 1000;
                    if (elapsed >= tile.crop.growTime) {
                        const cropDef = ITEMS[tile.crop.id];
                        gameState.inventory[tile.crop.id] = (gameState.inventory[tile.crop.id] || 0) + 2;
                        addExp(cropDef.exp || 3);
                        addCoins(1);

                        tile.crop = null;
                        playSFX('harvest');
                        playerRabbit.actionAnimTimer = 25;
                        showToast(`+2 ${cropDef.icon} ${cropDef.name}`, clickX, clickY);
                        hideTilePopover();
                    } else {
                        const remain = Math.ceil(tile.crop.growTime - elapsed);
                        showToast(`⏳ 成長中... あと${remain}秒`, clickX, clickY);
                        hideTilePopover();
                    }
                } else {
                    let html = `
                        <div class="text-xs font-black text-amber-950 mb-1">🌱 好きなタネをまく:</div>
                        <div class="flex space-x-1.5 mb-2">
                    `;
                    Object.values(ITEMS).filter(i => i.type === 'crop' && i.unlockLv <= gameState.level).forEach(seed => {
                        html += `
                            <button onclick="plantSeed(${r}, ${c}, '${seed.id}')" class="bg-amber-100 hover:bg-amber-200 border border-amber-300 p-1.5 rounded-xl flex flex-col items-center transition active:scale-95 min-w-[48px]">
                                <span class="text-base">${seed.icon}</span>
                                <span class="text-[9px] font-black text-amber-900">${seed.name}</span>
                            </button>
                        `;
                    });
                    html += `
                        </div>
                    `;
                    if (tile.building === 'plot') {
                        html += `
                            <button onclick="removeBuildingOrSoil(${r}, ${c})" class="w-full bg-slate-100 hover:bg-slate-200 text-slate-700 font-extrabold text-[10px] py-1 rounded-xl border border-slate-300 transition">
                                🧹 芝生に戻す (片付ける)
                            </button>
                        `;
                    }
                    showTilePopover(e.clientX, e.clientY, html);
                }
                return;
            }

            // Animal Buildings Popover
            if (tile.building === 'cow_barn') {
                handleAnimalTile(r, c, 'milk', 'wheat', '🐮 牛さんのお世話', '🥛 牛乳', e.clientX, e.clientY);
                return;
            }
            if (tile.building === 'chicken_coop') {
                handleAnimalTile(r, c, 'egg', 'wheat', '🐔 にわとりのお世話', '🥚 たまご', e.clientX, e.clientY);
                return;
            }
            if (tile.building === 'pig_pen') {
                handleAnimalTile(r, c, 'ham', 'carrot', '🐷 ぶたさんのお世話', '🥓 ハム', e.clientX, e.clientY);
                return;
            }

            // Bakery / Kitchen Facility Actions
            if (tile.building === 'bakery') {
                openModal('recipe-modal');
                hideTilePopover();
                return;
            }

            // Other decor buildings options
            if (tile.building) {
                showTilePopover(e.clientX, e.clientY, `
                    <div class="text-xs font-black text-amber-950 mb-1.5">${BUILDINGS[tile.building].name}</div>
                    <button onclick="removeBuildingOrSoil(${r}, ${c})" class="w-full bg-slate-100 hover:bg-slate-200 text-slate-700 font-extrabold text-[10px] py-1 rounded-xl border border-slate-300 transition">
                        🧹 施設を片付ける (撤去)
                    </button>
                `);
                return;
            }

            hideTilePopover();
        });

        function plantSeed(r, c, seedId) {
            const seedDef = ITEMS[seedId];
            const tile = gameState.grid[r][c];

            // Auto-create soil plot if it's empty grass
            tile.building = 'plot';
            tile.crop = {
                id: seedId,
                plantedAt: Date.now(),
                growTime: seedDef.growTime
            };
            playSFX('pop');
            showToast(`🌱 ${seedDef.name}を植えました！`, window.innerWidth/2, window.innerHeight/3);
            hideTilePopover();
        }

        function removeBuildingOrSoil(r, c) {
            const tile = gameState.grid[r][c];
            tile.building = null;
            tile.crop = null;
            tile.animal = null;
            playSFX('pop');
            showToast(`✨ きれいに片付けました`, window.innerWidth/2, window.innerHeight/3);
            hideTilePopover();
        }

        function buyItem(itemId, price) {
            if (gameState.coins >= price) {
                addCoins(-price);
                gameState.inventory[itemId] = (gameState.inventory[itemId] || 0) + 1;
                playSFX('coin');
                renderShopItems();
                showToast(`🛒 ${ITEMS[itemId].name}を買いました！`, window.innerWidth/2, window.innerHeight/3);
            } else {
                showToast(`⚠ コインが足りません`, window.innerWidth/2, window.innerHeight/3);
            }
        }

        }

            const list = document.getElementById('barn-items-list');

                        </div>

            }
        }

        function renderRecipes() {
            const list = document.getElementById('recipe-items-list');
            list.innerHTML = '';

            Object.values(ITEMS).filter(i => i.type === 'craft').forEach(craft => {
                const isUnlocked = craft.unlockLv <= gameState.level;
                
                let reqsHTML = '';
                let canCraft = true;

                Object.entries(craft.ingredients).forEach(([reqKey, reqQty]) => {
                    const have = gameState.inventory[reqKey] || 0;
                    if (have < reqQty) canCraft = false;
                    const reqDef = ITEMS[reqKey];
                    reqsHTML += `<span class="${have >= reqQty ? 'text-emerald-600' : 'text-rose-500'} font-bold">${reqDef ? reqDef.icon : ''}${reqQty} </span>`;
                });

                const div = document.createElement('div');
                div.className = "bg-white p-2.5 rounded-2xl border border-pink-200 flex items-center justify-between shadow-sm";
                div.innerHTML = `
                    <div class="flex items-center space-x-2.5">
                        <span class="text-2xl">${craft.icon}</span>
                        <div>
                            <div class="font-black text-xs text-amber-950">${craft.name}</div>
                            <div class="text-[10px] text-slate-600">必要: ${reqsHTML}</div>
                        </div>
                    </div>
                    <button onclick="craftItem('${craft.id}')" ${(!isUnlocked || !canCraft) ? 'disabled' : ''} 
                        class="px-3 py-1.5 rounded-xl font-black text-xs shadow transition ${isUnlocked && canCraft ? 'bg-gradient-to-r from-rose-400 to-pink-500 text-white active:scale-95' : 'bg-slate-200 text-slate-400'}">
                        ${isUnlocked ? '調理する' : `Lv.${craft.unlockLv}`}
                    </button>
                `;
                list.appendChild(div);
            });
        }

        function craftItem(craftId) {
            const craftDef = ITEMS[craftId];
            Object.entries(craftDef.ingredients).forEach(([reqKey, reqQty]) => {
                gameState.inventory[reqKey] -= reqQty;
            });

            gameState.inventory[craftId] = (gameState.inventory[craftId] || 0) + 1;
            addExp(craftDef.exp);
            playSFX('harvest');
            renderRecipes();
            showToast(`✨ ${craftDef.icon} ${craftDef.name}をつくりました！`, window.innerWidth/2, window.innerHeight/3);
        }
